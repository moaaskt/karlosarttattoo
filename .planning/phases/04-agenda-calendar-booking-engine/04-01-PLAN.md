/** Serializa qualquer Date para UTC ISO com ms e Z */
export function serializeDate(d: Date): string {
  return d.toISOString();
}

/** Valida se uma string é ISO UTC válida */
export function isValidISODate(s: string): boolean {
  return ISO_UTC_RE.test(s) && !isNaN(Date.parse(s));

**Funções a adicionar:**

#### `createBooking(data)` — com `BEGIN IMMEDIATE`

```ts
export function createBooking(data: CreateBookingInput): {
  booking?: Booking;
  error?: "time_block_conflict" | "booking_conflict" | "validation_error";
  warning?: "outside_hours";
  conflicts?: Array<{ id: string; client_name?: string; start_at: string; end_at: string }>;
};
```

Lógica interna:

1. Validar `start_at`, `end_at` com `isValidISODate` — erro `validation_error` se falhar.
2. Verificar `isWithinWorkHours` — se fora: incluir `warning: 'outside_hours'` na resposta (continua se `force: true`).
3. `db.exec('BEGIN IMMEDIATE')`:
   - Query time_blocks: `WHERE :start < end_at AND start_at < :end` — se hit: ROLLBACK + `time_block_conflict`.
   - Calcular `{ windowStart, windowEnd }` via `expandWindow(start, end, bufferMinutes)`.
   - Query bookings: `WHERE id != :selfId AND status NOT IN ('cancelado','no_show') AND :windowStart < end_at AND start_at < :windowEnd` — se hit: ROLLBACK + `booking_conflict`.
   - INSERT em bookings + INSERT em booking_events (`event_type: 'created'`).
4. COMMIT — retornar `{ booking }`.
